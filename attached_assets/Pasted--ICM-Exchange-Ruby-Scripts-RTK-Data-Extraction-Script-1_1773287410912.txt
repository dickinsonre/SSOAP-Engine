# ICM Exchange Ruby Scripts — RTK Data Extraction

## Script 1: Field Discovery — Find All RTK Fields in Your Network

Run this first to discover the exact field names in your version of ICM.

```ruby
# ============================================================================
# SCRIPT 1: RTK Field Discovery
# Purpose:  Scan the current network and list all RTK/RDII/UH-related fields
#           Works for both InfoWorks and SWMM networks
# Usage:    Open a network in ICM, run from the Ruby scripting console
# ============================================================================

net = WSApplication.current_network

puts "=" * 80
puts "RTK / RDII FIELD DISCOVERY"
puts "=" * 80
puts "Network: #{net.path}" rescue puts "Network path not available"
puts "Time: #{Time.now}"
puts ""

# ---------------------------------------------------------------------------
# Helper: scan a table for fields containing RTK-related keywords
# ---------------------------------------------------------------------------
def scan_table_for_rtk_fields(net, table_name, keywords)
  results = []
  begin
    objects = net.row_objects(table_name)
    if objects && objects.length > 0
      sample = objects[0]
      # Get all field names from the object
      fields = []
      begin
        fields = sample.fields rescue sample.table_info.fields rescue []
      rescue
        # Try alternate method to enumerate fields
      end
      
      # If we can enumerate fields, search for RTK-related ones
      if fields.respond_to?(:each)
        fields.each do |f|
          fname = f.respond_to?(:name) ? f.name : f.to_s
          keywords.each do |kw|
            if fname.downcase.include?(kw.downcase)
              val = sample[fname] rescue "N/A"
              results << { table: table_name, field: fname, sample_value: val }
            end
          end
        end
      end
      
      puts "  #{table_name}: #{objects.length} objects found"
    else
      puts "  #{table_name}: empty or not present"
    end
  rescue => e
    puts "  #{table_name}: not available (#{e.message})"
  end
  results
end

# ---------------------------------------------------------------------------
# Keywords that indicate RTK/RDII/Unit Hydrograph data
# ---------------------------------------------------------------------------
rtk_keywords = [
  'rtk', 'rdii', 'unit_hydrograph', 'hydrograph', 'uh_group',
  'sewer_area', 'contributing_area', 'rainfall_profile',
  'r1', 'r2', 'r3', 't1', 't2', 't3', 'k1', 'k2', 'k3',
  'r_short', 'r_medium', 'r_long',
  't_short', 't_medium', 't_long',
  'k_short', 'k_medium', 'k_long',
  'ia_max', 'ia_init', 'ia_recovery', 'abstraction',
  'max_depth_short', 'max_depth_medium', 'max_depth_long',
  'recovery_rate', 'starting_depth',
  'dmax', 'drec', 'd0'
]

# ---------------------------------------------------------------------------
# Tables to scan — InfoWorks and SWMM
# ---------------------------------------------------------------------------
iw_tables = [
  'hw_subcatchment', 'hw_node', 'hw_conduit',
  'hw_rtk_hydrograph', 'hw_unit_hydrograph',
  'subcatchment', 'node', 'conduit',
  '_subcatchments', '_nodes', '_links'
]

sw_tables = [
  'sw_subcatchment', 'sw_node', 'sw_junction', 'sw_outfall',
  'sw_conduit', 'sw_pump', 'sw_orifice', 'sw_weir',
  'sw_unit_hydrograph', 'sw_hydrograph', 'sw_rdii',
  'sw_storage'
]

all_tables = iw_tables + sw_tables

puts "Scanning InfoWorks tables..."
puts "-" * 40
all_rtk_fields = []

all_tables.each do |tbl|
  found = scan_table_for_rtk_fields(net, tbl, rtk_keywords)
  all_rtk_fields.concat(found)
end

puts ""
puts "=" * 80
puts "RTK-RELATED FIELDS FOUND"
puts "=" * 80

if all_rtk_fields.empty?
  puts "No RTK fields found automatically."
  puts "Try examining subcatchment objects manually:"
  puts ""
  puts "  net.row_objects('hw_subcatchment').each do |sc|"
  puts "    puts sc.id"
  puts "    # Try common field names:"
  puts "    ['r1','t1','k1','rtk_r1'].each do |f|"
  puts "      puts \"  #{f}: #{sc[f]}\" rescue nil"  
  puts "    end"
  puts "  end"
else
  all_rtk_fields.each do |r|
    puts "  Table: %-25s  Field: %-30s  Sample: %s" % [r[:table], r[:field], r[:sample_value]]
  end
end

puts ""
puts "=" * 80
puts "OBJECT COUNTS"
puts "=" * 80

count_tables = {
  'hw_subcatchment'     => 'InfoWorks Subcatchments',
  'hw_node'             => 'InfoWorks Nodes',
  'hw_conduit'          => 'InfoWorks Conduits',
  'sw_subcatchment'     => 'SWMM Subcatchments',
  'sw_node'             => 'SWMM Nodes',
  'sw_conduit'          => 'SWMM Conduits',
  'sw_unit_hydrograph'  => 'SWMM Unit Hydrographs'
}

count_tables.each do |tbl, label|
  begin
    objs = net.row_objects(tbl)
    count = objs ? objs.length : 0
    puts "  %-30s  %d objects" % [label, count] if count > 0
  rescue
    # table doesn't exist in this network type
  end
end

puts ""
puts "Discovery complete."
```

## Script 2: InfoWorks Network — Extract RTK from Subcatchments

```ruby
# ============================================================================
# SCRIPT 2: InfoWorks ICM — RTK Hydrograph Extraction
# Purpose:  Extract all RTK parameters from InfoWorks subcatchments
#           including monthly variations and initial abstraction
# Output:   CSV file + console summary
# ============================================================================

net = WSApplication.current_network

# ---------------------------------------------------------------------------
# Configuration — adjust output path as needed
# ---------------------------------------------------------------------------
OUTPUT_DIR   = 'C:/SSOAP/exports'         # Change to your preferred directory
TIMESTAMP    = Time.now.strftime('%Y%m%d_%H%M%S')
CSV_FILE     = "#{OUTPUT_DIR}/iw_rtk_export_#{TIMESTAMP}.csv"
INP_FILE     = "#{OUTPUT_DIR}/iw_rtk_hydrographs_#{TIMESTAMP}.inp"

# Create output directory if it doesn't exist
Dir.mkdir(OUTPUT_DIR) rescue nil

# ---------------------------------------------------------------------------
# RTK field name mapping — InfoWorks subcatchments
# These are the most common field names; adjust if discovery script
# reveals different names in your ICM version
# ---------------------------------------------------------------------------
RTK_FIELDS = {
  # Core identifiers
  id:                   'id',
  node_id:              'node_id',              # downstream node
  rtk_group:            'rtk_hydrograph',       # RTK hydrograph group name
  rainfall_profile:     'rainfall_profile',     # rain gage assignment
  contributing_area:    'contributing_area',     # sewer area (acres or ha)
  total_area:           'total_area',           # total subcatchment area
  
  # Short-term (fast) response — UH1
  r1:                   'r1',
  t1:                   't1',
  k1:                   'k1',
  ia_max_short:         'max_depth_short',      # initial abstraction max depth
  ia_recovery_short:    'recovery_rate_short',  # abstraction recovery rate
  ia_init_short:        'starting_depth_short', # initial depth used at start
  
  # Medium-term response — UH2
  r2:                   'r2',
  t2:                   't2',
  k2:                   'k2',
  ia_max_medium:        'max_depth_medium',
  ia_recovery_medium:   'recovery_rate_medium',
  ia_init_medium:       'starting_depth_medium',
  
  # Long-term (slow) response — UH3
  r3:                   'r3',
  t3:                   't3',
  k3:                   'k3',
  ia_max_long:          'max_depth_long',
  ia_recovery_long:     'recovery_rate_long',
  ia_init_long:         'starting_depth_long'
}

# ---------------------------------------------------------------------------
# Month names for seasonal RTK extraction
# ---------------------------------------------------------------------------
MONTHS = [
  'January', 'February', 'March', 'April', 'May', 'June',
  'July', 'August', 'September', 'October', 'November', 'December'
]

MONTH_ABBREV = [
  'JAN', 'FEB', 'MAR', 'APR', 'MAY', 'JUN',
  'JUL', 'AUG', 'SEP', 'OCT', 'NOV', 'DEC'
]

# ---------------------------------------------------------------------------
# Helper: safely read a field value, return nil if field doesn't exist
# ---------------------------------------------------------------------------
def safe_read(obj, field_name)
  begin
    val = obj[field_name]
    return val
  rescue
    return nil
  end
end

# ---------------------------------------------------------------------------
# Helper: try multiple possible field names, return first non-nil
# ---------------------------------------------------------------------------
def try_fields(obj, *field_names)
  field_names.each do |fn|
    val = safe_read(obj, fn)
    return val unless val.nil?
  end
  nil
end

# ---------------------------------------------------------------------------
# Extract RTK data from all subcatchments
# ---------------------------------------------------------------------------
puts "=" * 80
puts "INFOWORKS RTK HYDROGRAPH EXTRACTION"
puts "=" * 80

subcatchments = nil
table_name = nil

# Try different table names for InfoWorks subcatchments
['hw_subcatchment', 'subcatchment', '_subcatchments'].each do |tbl|
  begin
    subcatchments = net.row_objects(tbl)
    if subcatchments && subcatchments.length > 0
      table_name = tbl
      break
    end
  rescue
    next
  end
end

if subcatchments.nil? || subcatchments.length == 0
  puts "ERROR: No subcatchments found. Is this an InfoWorks network?"
  puts "       For SWMM networks, use Script 3 instead."
  exit
end

puts "Found #{subcatchments.length} subcatchments in table '#{table_name}'"
puts ""

# ---------------------------------------------------------------------------
# Collect RTK data
# ---------------------------------------------------------------------------
rtk_data = []
rtk_groups = {}
subcatchments_with_rtk = 0
subcatchments_without_rtk = 0

subcatchments.each do |sc|
  sc_id = sc.id
  
  # Read core identifiers
  node_id    = try_fields(sc, 'node_id', 'us_node_id', 'outlet_node', 'downstream_node')
  rtk_group  = try_fields(sc, 'rtk_hydrograph', 'rtk_group', 'unit_hydrograph',
                               'uh_group', 'rtk_hydrograph_id')
  rain_prof   = try_fields(sc, 'rainfall_profile', 'rain_gage', 'raingage')
  cont_area   = try_fields(sc, 'contributing_area', 'sewer_area', 'rdii_area')
  total_area  = try_fields(sc, 'total_area', 'area')
  
  # Read RTK parameters — try multiple possible field names
  r1 = try_fields(sc, 'r1', 'rtk_r1', 'r_short', 'short_r')
  t1 = try_fields(sc, 't1', 'rtk_t1', 't_short', 'short_t')
  k1 = try_fields(sc, 'k1', 'rtk_k1', 'k_short', 'short_k')
  
  r2 = try_fields(sc, 'r2', 'rtk_r2', 'r_medium', 'medium_r')
  t2 = try_fields(sc, 't2', 'rtk_t2', 't_medium', 'medium_t')
  k2 = try_fields(sc, 'k2', 'rtk_k2', 'k_medium', 'medium_k')
  
  r3 = try_fields(sc, 'r3', 'rtk_r3', 'r_long', 'long_r')
  t3 = try_fields(sc, 't3', 'rtk_t3', 't_long', 'long_t')
  k3 = try_fields(sc, 'k3', 'rtk_k3', 'k_long', 'long_k')
  
  # Read initial abstraction parameters
  ia_max_s = try_fields(sc, 'max_depth_short', 'ia_max_short', 'dmax1', 'ia_max_1')
  ia_max_m = try_fields(sc, 'max_depth_medium', 'ia_max_medium', 'dmax2', 'ia_max_2')
  ia_max_l = try_fields(sc, 'max_depth_long', 'ia_max_long', 'dmax3', 'ia_max_3')
  
  ia_rec_s = try_fields(sc, 'recovery_rate_short', 'ia_recovery_short', 'drec1', 'ia_rec_1')
  ia_rec_m = try_fields(sc, 'recovery_rate_medium', 'ia_recovery_medium', 'drec2', 'ia_rec_2')
  ia_rec_l = try_fields(sc, 'recovery_rate_long', 'ia_recovery_long', 'drec3', 'ia_rec_3')
  
  ia_init_s = try_fields(sc, 'starting_depth_short', 'ia_init_short', 'd01', 'ia_init_1')
  ia_init_m = try_fields(sc, 'starting_depth_medium', 'ia_init_medium', 'd02', 'ia_init_2')
  ia_init_l = try_fields(sc, 'starting_depth_long', 'ia_init_long', 'd03', 'ia_init_3')
  
  # Check if this subcatchment has RTK data
  has_rtk = (r1 && r1.to_f > 0) || (r2 && r2.to_f > 0) || (r3 && r3.to_f > 0) ||
            (rtk_group && rtk_group.to_s.strip.length > 0)
  
  if has_rtk
    subcatchments_with_rtk += 1
    
    r_total = (r1.to_f + r2.to_f + r3.to_f)
    
    record = {
      sc_id:          sc_id,
      node_id:        node_id,
      rtk_group:      rtk_group,
      rainfall_prof:  rain_prof,
      cont_area:      cont_area.to_f,
      total_area:     total_area.to_f,
      r1: r1.to_f, t1: t1.to_f, k1: k1.to_f,
      r2: r2.to_f, t2: t2.to_f, k2: k2.to_f,
      r3: r3.to_f, t3: t3.to_f, k3: k3.to_f,
      r_total:        r_total,
      ia_max_s: ia_max_s.to_f,  ia_max_m: ia_max_m.to_f,  ia_max_l: ia_max_l.to_f,
      ia_rec_s: ia_rec_s.to_f,  ia_rec_m: ia_rec_m.to_f,  ia_rec_l: ia_rec_l.to_f,
      ia_init_s: ia_init_s.to_f, ia_init_m: ia_init_m.to_f, ia_init_l: ia_init_l.to_f
    }
    
    rtk_data << record
    
    # Track unique RTK groups
    grp = rtk_group.to_s.strip
    rtk_groups[grp] = [] unless rtk_groups.key?(grp)
    rtk_groups[grp] << sc_id
  else
    subcatchments_without_rtk += 1
  end
end

# ---------------------------------------------------------------------------
# Try to extract monthly RTK variations
# ---------------------------------------------------------------------------
monthly_rtk = {}

subcatchments.each do |sc|
  sc_id = sc.id
  has_monthly = false
  
  MONTH_ABBREV.each_with_index do |mon, mi|
    month_num = mi + 1
    
    # Try various naming conventions for monthly RTK
    r1m = try_fields(sc, "r1_#{mon.downcase}", "r1_m#{month_num}", "#{mon.downcase}_r1")
    t1m = try_fields(sc, "t1_#{mon.downcase}", "t1_m#{month_num}", "#{mon.downcase}_t1")
    k1m = try_fields(sc, "k1_#{mon.downcase}", "k1_m#{month_num}", "#{mon.downcase}_k1")
    
    r2m = try_fields(sc, "r2_#{mon.downcase}", "r2_m#{month_num}", "#{mon.downcase}_r2")
    t2m = try_fields(sc, "t2_#{mon.downcase}", "t2_m#{month_num}", "#{mon.downcase}_t2")
    k2m = try_fields(sc, "k2_#{mon.downcase}", "k2_m#{month_num}", "#{mon.downcase}_k2")
    
    r3m = try_fields(sc, "r3_#{mon.downcase}", "r3_m#{month_num}", "#{mon.downcase}_r3")
    t3m = try_fields(sc, "t3_#{mon.downcase}", "t3_m#{month_num}", "#{mon.downcase}_t3")
    k3m = try_fields(sc, "k3_#{mon.downcase}", "k3_m#{month_num}", "#{mon.downcase}_k3")
    
    if r1m || r2m || r3m
      has_monthly = true
      monthly_rtk[sc_id] = {} unless monthly_rtk.key?(sc_id)
      monthly_rtk[sc_id][mon] = {
        r1: r1m.to_f, t1: t1m.to_f, k1: k1m.to_f,
        r2: r2m.to_f, t2: t2m.to_f, k2: k2m.to_f,
        r3: r3m.to_f, t3: t3m.to_f, k3: k3m.to_f
      }
    end
  end
end

# ---------------------------------------------------------------------------
# Console Summary
# ---------------------------------------------------------------------------
puts "RESULTS SUMMARY"
puts "-" * 60
puts "  Total subcatchments:          #{subcatchments.length}"
puts "  With RTK data:                #{subcatchments_with_rtk}"
puts "  Without RTK data:             #{subcatchments_without_rtk}"
puts "  Unique RTK groups:            #{rtk_groups.keys.length}"
puts "  With monthly variation:       #{monthly_rtk.keys.length}"
puts ""

puts "RTK GROUP SUMMARY"
puts "-" * 60
rtk_groups.each do |grp, members|
  puts "  Group '#{grp}': #{members.length} subcatchments"
  puts "    Members: #{members.join(', ')}"
end
puts ""

puts "RTK PARAMETER SUMMARY"
puts "-" * 60
puts "  %-12s  %-8s %-8s %-8s %-8s %-8s %-8s %-8s %-8s %-8s %-8s" %
     ['Subcat', 'R1', 'T1', 'K1', 'R2', 'T2', 'K2', 'R3', 'T3', 'K3', 'R_total']
puts "  " + "-" * 100

rtk_data.each do |r|
  puts "  %-12s  %-8.4f %-8.2f %-8.2f %-8.4f %-8.2f %-8.2f %-8.4f %-8.2f %-8.2f %-8.4f" %
       [r[:sc_id], r[:r1], r[:t1], r[:k1], r[:r2], r[:t2], r[:k2],
        r[:r3], r[:t3], r[:k3], r[:r_total]]
end

# ---------------------------------------------------------------------------
# Statistics
# ---------------------------------------------------------------------------
if rtk_data.length > 0
  puts ""
  puts "RTK STATISTICS"
  puts "-" * 60
  
  [:r1, :r2, :r3, :r_total, :t1, :t2, :t3, :k1, :k2, :k3].each do |param|
    values = rtk_data.map { |r| r[param] }.select { |v| v > 0 }
    next if values.empty?
    
    avg = values.reduce(:+) / values.length
    min_v = values.min
    max_v = values.max
    sorted = values.sort
    median = sorted.length.odd? ? sorted[sorted.length/2] : 
             (sorted[sorted.length/2 - 1] + sorted[sorted.length/2]) / 2.0
    
    puts "  %-10s  min=%-10.4f  max=%-10.4f  mean=%-10.4f  median=%-10.4f  n=%d" %
         [param, min_v, max_v, avg, median, values.length]
  end
  
  # R-value severity classification
  puts ""
  puts "R-VALUE SEVERITY CLASSIFICATION (EPA SSOAP)"
  puts "-" * 60
  r_totals = rtk_data.map { |r| r[:r_total] * 100 }
  low      = r_totals.count { |r| r < 3 }
  moderate = r_totals.count { |r| r >= 3 && r < 6 }
  signif   = r_totals.count { |r| r >= 6 && r < 10 }
  severe   = r_totals.count { |r| r >= 10 }
  
  puts "  Low (R < 3%%):            %d subcatchments" % low
  puts "  Moderate (3-6%%):          %d subcatchments" % moderate
  puts "  Significant (6-10%%):      %d subcatchments" % signif
  puts "  Severe (R > 10%%):         %d subcatchments" % severe
end

# ---------------------------------------------------------------------------
# Export to CSV
# ---------------------------------------------------------------------------
begin
  File.open(CSV_FILE, 'w') do |f|
    # Header
    f.puts [
      'Subcatchment_ID', 'Node_ID', 'RTK_Group', 'Rainfall_Profile',
      'Contributing_Area', 'Total_Area',
      'R1', 'T1_hr', 'K1', 'R2', 'T2_hr', 'K2', 'R3', 'T3_hr', 'K3',
      'R_Total',
      'IA_Max_Short', 'IA_Max_Medium', 'IA_Max_Long',
      'IA_Recovery_Short', 'IA_Recovery_Medium', 'IA_Recovery_Long',
      'IA_Init_Short', 'IA_Init_Medium', 'IA_Init_Long'
    ].join(',')
    
    # Data rows
    rtk_data.each do |r|
      f.puts [
        r[:sc_id], r[:node_id], r[:rtk_group], r[:rainfall_prof],
        r[:cont_area], r[:total_area],
        r[:r1], r[:t1], r[:k1], r[:r2], r[:t2], r[:k2], r[:r3], r[:t3], r[:k3],
        r[:r_total],
        r[:ia_max_s], r[:ia_max_m], r[:ia_max_l],
        r[:ia_rec_s], r[:ia_rec_m], r[:ia_rec_l],
        r[:ia_init_s], r[:ia_init_m], r[:ia_init_l]
      ].join(',')
    end
  end
  puts ""
  puts "CSV exported to: #{CSV_FILE}"
rescue => e
  puts "CSV export error: #{e.message}"
end

# ---------------------------------------------------------------------------
# Export SWMM5 INP sections: [HYDROGRAPHS] and [RDII]
# ---------------------------------------------------------------------------
begin
  File.open(INP_FILE, 'w') do |f|
    f.puts ";; =================================================================="
    f.puts ";; SWMM5 RTK Hydrograph Data"
    f.puts ";; Extracted from InfoWorks ICM network"
    f.puts ";; Date: #{Time.now}"
    f.puts ";; Subcatchments with RTK: #{subcatchments_with_rtk}"
    f.puts ";; =================================================================="
    f.puts ""
    
    # ---- [HYDROGRAPHS] section ----
    f.puts "[HYDROGRAPHS]"
    f.puts ";;Name           Month      Response   R        T        K        " +
           "Dmax     Drec     D0"
    f.puts ";;-------------- ---------- ---------- -------- -------- -------- " +
           "-------- -------- --------"
    
    # Track which groups we've already written
    written_groups = {}
    
    rtk_data.each do |r|
      grp = r[:rtk_group].to_s.strip
      grp = "UH_#{r[:sc_id]}" if grp.empty?
      
      next if written_groups.key?(grp)
      written_groups[grp] = true
      
      rain = r[:rainfall_prof].to_s.strip
      rain = "*" if rain.empty?
      
      # Check for monthly variations
      if monthly_rtk.key?(r[:sc_id])
        # Write monthly parameters
        MONTH_ABBREV.each_with_index do |mon, mi|
          if monthly_rtk[r[:sc_id]] && monthly_rtk[r[:sc_id]][mon]
            m = monthly_rtk[r[:sc_id]][mon]
            f.puts "%-16s %-10s SHORT      %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
                   [grp, mon, m[:r1], m[:t1], m[:k1], r[:ia_max_s], r[:ia_rec_s], r[:ia_init_s]]
            f.puts "%-16s %-10s MEDIUM     %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
                   [grp, mon, m[:r2], m[:t2], m[:k2], r[:ia_max_m], r[:ia_rec_m], r[:ia_init_m]]
            f.puts "%-16s %-10s LONG       %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
                   [grp, mon, m[:r3], m[:t3], m[:k3], r[:ia_max_l], r[:ia_rec_l], r[:ia_init_l]]
          end
        end
      else
        # Write ALL (non-seasonal) parameters
        f.puts "%-16s %-10s SHORT      %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
               [grp, 'ALL', r[:r1], r[:t1], r[:k1], r[:ia_max_s], r[:ia_rec_s], r[:ia_init_s]]
        f.puts "%-16s %-10s MEDIUM     %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
               [grp, 'ALL', r[:r2], r[:t2], r[:k2], r[:ia_max_m], r[:ia_rec_m], r[:ia_init_m]]
        f.puts "%-16s %-10s LONG       %-8.4f %-8.2f %-8.2f %-8.4f %-8.4f %-8.4f" %
               [grp, 'ALL', r[:r3], r[:t3], r[:k3], r[:ia_max_l], r[:ia_rec_l], r[:ia_init_l]]
      end
      f.puts ""
    end
    
    # ---- [RDII] section ----
    f.puts ""
    f.puts "[RDII]"
    f.puts ";;Node           UHGroup          SewerArea"
    f.puts ";;-------------- ---------------- ---------"
    
    rtk_data.each do |r|
      node = r[:node_id].to_s.strip
```

```ruby
def to_json_value(val)
  case val
  when nil    then 'null'
  when String then '"' + val.gsub('"', '\\"') + '"'
  when Symbol then '"' + val.to_s + '"'
  when Numeric then val.to_s
  when true, false then val.to_s
  when Array  then '[' + val.map { |v| to_json_value(v) }.join(', ') + ']'
  when Hash   then to_json_hash(val)
  else '"' + val.to_s + '"'
  end
end

def to_json_hash(hash, indent = 0)
  pad = '  ' * indent
  inner_pad = '  ' * (indent + 1)
  lines = []
  hash.each do |k, v|
    if v.is_a?(Hash)
      lines << "#{inner_pad}#{to_json_value(k.to_s)}: #{to_json_hash(v, indent + 1)}"
    elsif v.is_a?(Array) && v.length > 0 && v[0].is_a?(Hash)
      arr_lines = v.map { |item| to_json_hash(item, indent + 2) }
      lines << "#{inner_pad}#{to_json_value(k.to_s)}: [\n#{arr_lines.join(",\n")}\n#{inner_pad}]"
    else
      lines << "#{inner_pad}#{to_json_value(k.to_s)}: #{to_json_value(v)}"
    end
  end
  "{\n#{lines.join(",\n")}\n#{pad}}"
end

begin
```

## Script 6: Write RTK Back to ICM Network (Import)

```ruby
# ============================================================================
# SCRIPT 6: Import RTK Parameters into ICM Network
# Purpose:  Read RTK parameters from CSV and write them to network objects
#           Works for both InfoWorks and SWMM networks
# Usage:    Prepare a CSV with columns matching the format below, then run
# ============================================================================

net = WSApplication.current_network

# ---------------------------------------------------------------------------
# Configuration — set your input CSV path
# ---------------------------------------------------------------------------
INPUT_CSV = 'C:/SSOAP/exports/calibrated_rtk_params.csv'

# Expected CSV format:
# For InfoWorks: Subcatchment_ID,R1,T1,K1,R2,T2,K2,R3,T3,K3,IA_Max_S,IA_Rec_S,IA_Init_S,...

puts "=" * 80
puts "RTK PARAMETER IMPORT"
puts "=" * 80
puts "Input: #{INPUT_CSV}"
puts ""

# ---------------------------------------------------------------------------
# Read CSV

unless File.exist?(INPUT_CSV)
  puts "ERROR: File not found: #{INPUT_CSV}"
  puts "Create a CSV with calibrated RTK parameters first."
  exit
end

lines = File.readlines(INPUT_CSV).map(&:strip).reject(&:empty?)
header = lines[0].split(',').map(&:strip)
data_rows = lines[1..-1].map { |l| l.split(',').map(&:strip) }

puts "CSV columns: #{header.join(', ')}"
puts "Data rows: #{data_rows.length}"
puts ""

# ---------------------------------------------------------------------------
# Map column indices
# ---------------------------------------------------------------------------
def col_index(header, *possible_names)
  possible_names.each do |name|
    idx = header.index { |h| h.downcase == name.downcase }
    return idx if idx
  end
  nil
end

id_col   = col_index(header, 'Subcatchment_ID', 'Node_ID', 'ID', 'Name')
r1_col   = col_index(header, 'R1')
t1_col   = col_index(header, 'T1', 'T1_hr')
k1_col   = col_index(header, 'K1')
r2_col   = col_index(header, 'R2')
t2_col   = col_index(header, 'T2', 'T2_hr')
k2_col   = col_index(header, 'K2')
r3_col   = col_index(header, 'R3')
t3_col   = col_index(header, 'T3', 'T3_hr')
k3_col   = col_index(header, 'K3')
uh_col   = col_index(header, 'UH_Group', 'RTK_Group')
area_col = col_index(header, 'Sewer_Area', 'Contributing_Area', 'Area')

unless id_col && r1_col

  exit
end

# ---------------------------------------------------------------------------
# Determine target network type and apply
# ---------------------------------------------------------------------------
def try_write(obj, field_name, value)
  begin
    obj[field_name] = value
    return true
  rescue
    return false
  end
end

updated = 0
not_found = 0
errors = 0

# Try InfoWorks subcatchments first
target_table = nil
['hw_subcatchment', 'subcatchment'].each do |tbl|
  begin
    objs = net.row_objects(tbl)
    if objs && objs.length > 0

      break
    end
  rescue; end

  ['sw_node', 'sw_junction'].each do |tbl|

      objs = net.row_objects(tbl)
      if objs && objs.length > 0

    rescue; end

if target_table.nil?

end

puts "Target table: #{target_table}"
puts ""

# Build lookup of network objects by ID
objects_by_id = {}
net.row_objects(target_table).each do |obj|

end

# ---------------------------------------------------------------------------

net.transaction_begin

data_rows.each_with_index do |row, idx|

  unless obj

    ['k1', 'rtk_k1', 'k_short'].each { |f| try_write(obj, f, k1) } if k1_col
    ['r2', 'rtk_r2', 'r_medium'].each { |f| try_write(obj, f, r2) } if r2_col
    ['t2', 'rtk_t2', 't_medium'].each { |f| try_write(obj, f, t2) } if t2_col
    ['k2', 'rtk_k2', 'k_medium'].each { |f| try_write(obj, f, k2) } if k2_col

    # Write UH group and area if present
    if uh_col && row[uh_col]
      ['rtk_hydrograph', 'unit_hydrograph', 'uh_group'].each { |f| try_write(obj, f, row[uh_col]) }
    end
    if area_col && row[area_col]
      ['contributing_area', 'sewer_area', 'rdii_sewer_area'].each { |f| try_write(obj, f, row[area_col].to_f) }
    end
    
    obj.write
    
    r_total = (r1 || 0) + (r2 || 0) + (r3 || 0)
    puts "  ✓ #{obj_id}: R=%.4f T=[%.1f,%.1f,%.1f] K=[%.1f,%.1f,%.1f]" %
         [r_total, t1||0, t2||0, t3||0, k1||0, k2||0, k3||0]
    updated += 1
    
  rescue => e
    puts "  ✗ #{obj_id}: ERROR — #{e.message}"
    errors += 1
  end
end

net.transaction_commit

# ---------------------------------------------------------------------------
# Summary
# ---------------------------------------------------------------------------
puts ""
puts "=" * 80
puts "IMPORT COMPLETE"
puts "=" * 80
puts "  Updated:    #{updated}"
puts "  Not found:  #{not_found}"
puts "  Errors:     #{errors}"
puts ""

if errors == 0 && not_found == 0
  puts "  ✓ All parameters imported successfully"
elsif errors > 0
  puts "  ⚠ Some errors occurred — review the log above"
  puts "    The transaction was still committed for successful rows."
  puts "    Run the extraction script (Script 2 or 3) to verify."
end
```
